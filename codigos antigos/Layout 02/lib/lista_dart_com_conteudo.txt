import 'package:flutter/material.dart';
import 'package:cloud_firestore/cloud_firestore.dart';

class ConfigurationProvider with ChangeNotifier {
  Map<String, dynamic>? _providerConfig;
  bool _isLoading = true;
  String? _errorMessage;

  Map<String, dynamic>? get providerConfig => _providerConfig;
  bool get isLoading => _isLoading;
  String? get errorMessage => _errorMessage;

  Future<void> loadConfig(String providerId) async {
    _isLoading = true;
    _errorMessage = null;
    notifyListeners();

    try {
      final doc = await FirebaseFirestore.instance
          .collection('provedores')
          .doc(providerId)
          .get();

      if (doc.exists) {
        _providerConfig = doc.data();
        print("✅ Configuração carregada e disponível globalmente via Provider.");
      } else {
        _errorMessage = "Provedor com ID '$providerId' não encontrado.";
        print("❌ ERRO FATAL: $_errorMessage");
      }
    } catch (e) {
      _errorMessage = "Erro de conexão ao buscar configuração: $e";
      print("❌ ERRO FATAL: $_errorMessage");
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }
}

// Função auxiliar para converter a cor que vem do Firebase
Color _hexToColor(String hexString) {
  final buffer = StringBuffer();
  if (hexString.length == 6 || hexString.length == 7) buffer.write('ff');
  buffer.write(hexString.replaceAll('#', ''));
  return Color(int.parse(buffer.toString(), radix: 16));
}

class ConsumoPage extends StatefulWidget {

class _ConsumoPageState extends State<ConsumoPage> {
  bool _isLoading = true;
  String _errorMessage = '';
  double _usedGb = 0;
  String _planName = '';
  String _period = '';

  late final Color primaryColor;

  @override
  void initState() {
    super.initState();
    final configMap = widget.providerConfig['config'] as Map<String, dynamic>? ?? {};

    primaryColor = hexToColor(configMap['themeColor'] as String? ?? '#673AB7');

    _fetchConsumptionData();
  }

  Future<void> _fetchConsumptionData() async {
    const apiUrl = 'http://45.176.56.70:3000/get-consumption-data';

    try {
      final details = widget.providerConfig['details'] as Map<String, dynamic>? ?? {};
      final sgpParams = {
        "token": details['apiToken'] as String? ?? '',
        "app": details['appName'] as String? ?? ''
      };
      if ((sgpParams['token'] ?? '').isEmpty || (sgpParams['app'] ?? '').isEmpty) {
        throw Exception("Token ou App Name não configurados no painel.");
      }

      final response = await http.post(
        Uri.parse(apiUrl),
        headers: {'Content-Type': 'application/json'},
        body: json.encode({
          'cpfCnpj': widget.cpfCnpj,
          'senha': widget.senha,
          'sgpParams': sgpParams,
        }),
      );

      if (mounted) {
        if (response.statusCode == 200) {
          final responseData = json.decode(response.body)['data'];
          setState(() {
            // ===== CORREÇÃO APLICADA AQUI PARA EVITAR ERRO DE NULO =====
            _usedGb = (responseData['usedGb'] as num? ?? 0).toDouble();
            _planName = responseData['planName'] as String? ?? 'Plano não informado';
            _period = responseData['period'] as String? ?? 'N/A';
            _isLoading = false;
            // ==========================================================
          });
        } else {
          final errorBody = json.decode(response.body)['error'];
          setState(() {
            _errorMessage = errorBody['message'] ?? 'Não foi possível carregar os dados.';
            _isLoading = false;
          });
        }
      }
    } catch (e) {
      if (mounted) {
        setState(() {
          _errorMessage = 'Erro de conexão: ${e.toString()}';
          _isLoading = false;
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFF3F4F8),
      appBar: AppBar(
        title: Text('Meu Consumo', style: TextStyle(color: _isLoading ? Colors.grey.shade800 : primaryColor, fontWeight: FontWeight.bold)),

    );
  }

    _autoScrollTimer = Timer.periodic(const Duration(seconds: 5), (timer) {
      if (widget.imageUrls.isEmpty || !mounted || !_pageController.hasClients) return;
      int nextPage = _pageController.page!.round() + 1;
      if (nextPage >= widget.imageUrls.length) {
        nextPage = 0;
      }
      _pageController.animateToPage(

        curve: Curves.easeOutCubic,

class MeuIpPage extends StatefulWidget {
  final Map<String, dynamic> providerConfig;

  const MeuIpPage({Key? key, required this.providerConfig}) : super(key: key);

  @override
  State<MeuIpPage> createState() => _MeuIpPageState();
}

class _MeuIpPageState extends State<MeuIpPage> {
  bool _isLoading = true;
  String? _errorMessage;
  Map<String, dynamic>? _ipData;
  late final Color primaryColor;

  @override
  void initState() {
    super.initState();
    final configMap = widget.providerConfig['config'] as Map<String, dynamic>? ?? {};

    // ===== CORREÇÃO APLICADA AQUI =====
    primaryColor = hexToColor(configMap['themeColor'] as String? ?? '#673AB7');
    // ===================================

    _fetchIpInfo();
  }

  Future<void> _fetchIpInfo() async {
    try {
      final response = await http.get(Uri.parse('https://ipinfo.io/json'));
      if (mounted) {
        if (response.statusCode == 200) {
          setState(() { _ipData = json.decode(response.body); _isLoading = false; });
        } else {
          setState(() { _errorMessage = 'Não foi possível obter os dados de IP.'; _isLoading = false; });
        }
      }
    } catch (e) {
      if (mounted) {
        setState(() { _errorMessage = 'Erro de conexão. Verifique a sua internet.'; _isLoading = false; });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      backgroundColor: const Color(0xFFF3F4F8),
      appBar: AppBar(
        title: Text('Meu Endereço IP', style: TextStyle(color: primaryColor, fontWeight: FontWeight.bold)),
        backgroundColor: Colors.transparent,
        elevation: 0,
        iconTheme: IconThemeData(color: primaryColor),
      ),
      body: Center(child: Padding(padding: const EdgeInsets.all(20.0), child: _buildContent())),
    );
  }

  Widget _buildContent() {
    if (_isLoading) return _buildShimmerLoading();
    if (_errorMessage != null) return Text(_errorMessage!, style: const TextStyle(fontSize: 16, color: Colors.red));
    if (_ipData != null) return _buildIpInfoCard();
    return const SizedBox.shrink();
  }

  Widget _buildIpInfoCard() {
    return Card(
      elevation: 4,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
      child: Padding(
        padding: const EdgeInsets.all(24.0),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            const FaIcon(FontAwesomeIcons.networkWired, size: 40, color: Colors.grey),
            const SizedBox(height: 16),
            Text("Seu IP Público é:", style: TextStyle(fontSize: 16, color: Colors.grey.shade600)),
            const SizedBox(height: 8),
            Text(_ipData!['ip'] ?? 'Não encontrado', style: TextStyle(fontSize: 28, fontWeight: FontWeight.bold, color: primaryColor)),
            const Divider(height: 40),
            _buildInfoRow(icon: Icons.location_city, title: "Localização", value: "${_ipData!['city'] ?? ''}, ${_ipData!['region'] ?? ''}"),
            const SizedBox(height: 12),
            _buildInfoRow(icon: Icons.router, title: "Provedor", value: _ipData!['org'] ?? 'Não encontrado'),
          ],
        ),
      ),
    );
  }

  Widget _buildInfoRow({required IconData icon, required String title, required String value}) {
    return Row(
      children: [
        Icon(icon, color: Colors.grey.shade500, size: 20),
        const SizedBox(width: 16),
        Text("$title:", style: const TextStyle(fontWeight: FontWeight.bold)),
        const SizedBox(width: 8),
        Expanded(child: Text(value, textAlign: TextAlign.end, style: TextStyle(color: Colors.grey.shade700))),
      ],
    );
  }

  Widget _buildShimmerLoading() {
    return Shimmer.fromColors(
      baseColor: Colors.grey.shade300,
      highlightColor: Colors.grey.shade100,
      child: Card(
        elevation: 4,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
        child: const Padding(
          padding: EdgeInsets.all(24.0),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              SkeletonContainer(width: 40, height: 40, borderRadius: BorderRadius.all(Radius.circular(20))),
              SizedBox(height: 16),
              SkeletonContainer(width: 120, height: 16),
              SizedBox(height: 8),
              SkeletonContainer(width: 200, height: 28),
              Divider(height: 40),
              SkeletonContainer(height: 20),
              SizedBox(height: 12),
              SkeletonContainer(height: 20),
            ],
          ),
        ),
      ),
    );
  }
}

void navigateToPageById({
  required BuildContext context,
  required String pageId,
  required Map<String, dynamic> providerConfig,
  required String cpfCnpj,
  required String senha,
  required String status,
}) {
  final configMap = providerConfig['config'] as Map<String, dynamic>? ?? {};
  final suporteMap = configMap['support'] as Map<String, dynamic>? ?? {};
  final whatsappNumber = suporteMap['whatsapp'] as String? ?? '';

  void _push(Widget page) {
    Navigator.push(context, MaterialPageRoute(builder: (context) => page));
  }

  void _launch(String url) async {
    final uri = Uri.parse(url);
    if (!await launchUrl(uri, mode: LaunchMode.externalApplication)) {
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Não foi possível abrir o link: $url')));
    }
  }

  switch (pageId) {
    case 'internet_usage':
      _push(ConsumoPage(providerConfig: providerConfig, cpfCnpj: cpfCnpj, senha: senha));
      break;

  // ===== CORREÇÃO APLICADA AQUI =====
    case 'support':
      _push(SuportePage(
        providerConfig: providerConfig,
        cpfCnpj: cpfCnpj,
        senha: senha,
        status: status,
      ));
      break;
  // ===================================

    case 'invoices':
      _push(FinanceiroPage(providerConfig: providerConfig, cpfCnpj: cpfCnpj, senha: senha));
      break;
    case 'speed_test':
      _push(SpeedTestPage(providerConfig: providerConfig));
      break;
    case 'my_ip':
      _push(MeuIpPage(providerConfig: providerConfig));
      break;
    case 'useful_tips':
      _push(DicasUteisPage(providerConfig: providerConfig));
      break;
    case 'faq':
      _push(FaqPage(providerConfig: providerConfig));
      break;
    case 'notifications':
      _push(NotificacoesPage(providerConfig: providerConfig));
      break;
    case 'payment_promise':
      _push(PromessaPage(status: status, cpfCnpj: cpfCnpj, providerConfig: providerConfig));
      break;
    case 'whatsapp':
      _launch('https://wa.me/$whatsappNumber');
      break;
    case 'down_detector':
      _push(DownDetectorPage(providerConfig: providerConfig));
      break;
    default:
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Funcionalidade "$pageId" não implementada.')));
  }
}

class PainelPage extends StatefulWidget {
  final String cpfCnpj, senha, content, plano, status, valorFatura, vencimentoFatura;
  final Map<String, dynamic> providerConfig;

  const PainelPage({
    Key? key,
    required this.cpfCnpj, required this.senha, required this.content, required this.plano,
    required this.status, required this.valorFatura, required this.vencimentoFatura,
    required this.providerConfig,
  }) : super(key: key);

  @override
  _PainelPageState createState() => _PainelPageState();
}

class _PainelPageState extends State<PainelPage> {
  int _selectedIndex = 0;
  List<Widget> _pages = [];
  List<MenuItem> _menuItems = [];

  late Color primaryColor;
  late Color secondaryColor;
  late String logoUrl;
  late Map<String, dynamic> funcionalidades;

  @override
  void initState() {
    super.initState();
    final configMap = widget.providerConfig['config'] as Map<String, dynamic>? ?? {};

    primaryColor = hexToColor(configMap['themeColor'] as String? ?? '#673AB7');
    secondaryColor = hexToColor(configMap['secondaryColor'] as String? ?? '#9575CD');

    logoUrl = configMap['logoUrl'] as String? ?? '';
    funcionalidades = configMap['features'] as Map<String, dynamic>? ?? {};

    _menuItems = _loadMenuConfig();
    _pages = _buildPages();
  }

  List<MenuItem> _parseMenuData(Map<String, dynamic> menuData) {
    final order = menuData['order'] as List<dynamic>? ?? [];
    final items = menuData['items'] as Map<String, dynamic>? ?? {};
    final List<MenuItem> enabledItems = [];
    for (var id in order) {
      final itemData = items[id.toString()] as Map<String, dynamic>?;
      if (itemData != null && (itemData['enabled'] as bool? ?? false)) {
        final type = itemData['type'] as String? ?? 'internal';

        enabledItems.add(MenuItem(
          id: id.toString(),
          name: itemData['name'] as String? ?? pageNames[id.toString()] ?? 'Menu',
          type: type,
          url: itemData['url'] as String?,
          icon: pageIcons[type == 'external_link' ? 'external_link' : id.toString()] ?? pageIcons['default']!,
        ));
      }
    }
    return enabledItems;
  }

  List<MenuItem> _loadMenuConfig() {
    final configMap = widget.providerConfig['config'] as Map<String, dynamic>?;
    final menuConfig = configMap?['menuConfig'] as Map<String, dynamic>?;

    if (menuConfig != null) {
      return _parseMenuData(menuConfig);
    }
    final rootMenuConfig = widget.providerConfig['menuConfig'] as Map<String, dynamic>?;
    if (rootMenuConfig != null) {
      return _parseMenuData(rootMenuConfig);
    }
    return [];
  }

  List<Widget> _buildPages() {
    final List<Widget> pageList = [];
    pageList.add(InicioTab(
      cpfCnpj: widget.cpfCnpj, senha: widget.senha, content: widget.content,
      plano: widget.plano, status: widget.status, valorFatura: widget.valorFatura,
      vencimentoFatura: widget.vencimentoFatura, menuItems: _menuItems,
      providerConfig: widget.providerConfig,
    ));
    if (funcionalidades['consumption'] as bool? ?? true) {
      pageList.add(ConsumoPage(cpfCnpj: widget.cpfCnpj, senha: widget.senha, providerConfig: widget.providerConfig));
    }
    if (funcionalidades['support'] as bool? ?? true) {
      pageList.add(SuportePage(
        providerConfig: widget.providerConfig,
        cpfCnpj: widget.cpfCnpj,
        senha: widget.senha,
        status: widget.status,
      ));
    }
    return pageList;
  }

  void _onItemTapped(int index) {
    setState(() { _selectedIndex = index; });
  }

  void _handleMenuNavigation(MenuItem item) async {
    Navigator.pop(context);
    if (item.type == 'external_link' && item.url != null) {
      final uri = Uri.parse(item.url!);
      if (await canLaunchUrl(uri)) {
        await launchUrl(uri, mode: LaunchMode.externalApplication);
      } else {
        ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Não foi possível abrir o link: ${item.url}')));
      }
    } else {
      navigateToPageById(
        context: context,
        pageId: item.id,
        cpfCnpj: widget.cpfCnpj,
        senha: widget.senha,
        status: widget.status,
        providerConfig: widget.providerConfig,
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    final navItems = _buildBottomNavItems(funcionalidades);
    return Scaffold(
      extendBody: true,
      backgroundColor: const Color(0xFFF8F7FF),
      drawer: _buildDrawer(funcionalidades, primaryColor, secondaryColor, logoUrl, _menuItems),
      bottomNavigationBar: navItems.isNotEmpty
          ? CustomBottomNavigationBar(
        selectedIndex: _selectedIndex,
        onItemTapped: _onItemTapped,
        primaryColor: primaryColor,
        items: navItems,
      )
          : null,
      body: _pages.length > _selectedIndex ? _pages[_selectedIndex] : const Center(child: Text("Módulo indisponível")),
    );
  }

  List<BottomNavigationBarItem> _buildBottomNavItems(Map<String, dynamic> funcs) {
    List<BottomNavigationBarItem> items = [];
    items.add(const BottomNavigationBarItem(icon: Icon(Icons.home_outlined), activeIcon: Icon(Icons.home), label: 'INÍCIO'));
    if (funcs['consumption'] as bool? ?? true) {
      items.add(const BottomNavigationBarItem(icon: Icon(Icons.data_usage_outlined), activeIcon: Icon(Icons.data_usage), label: 'CONSUMO'));
    }
    if (funcs['support'] as bool? ?? true) {
      items.add(const BottomNavigationBarItem(icon: Icon(Icons.support_agent_outlined), activeIcon: Icon(Icons.support_agent), label: 'SUPORTE'));
    }
    return items;
  }

  Drawer _buildDrawer(Map<String, dynamic> funcs, Color pColor, Color sColor, String lUrl, List<MenuItem> mItems) {
    int consumoIndex = (funcs['consumption'] as bool? ?? true) ? 1 : -1;
    int suporteIndex = (funcs['support'] as bool? ?? true) ? (consumoIndex != -1 ? 2 : 1) : -1;

    return Drawer(
      child: Container(
        color: Colors.white,
        child: Column(
          children: [
            DrawerHeader(
              margin: EdgeInsets.zero,
              padding: const EdgeInsets.fromLTRB(24, 20, 24, 20),
              decoration: BoxDecoration(gradient: LinearGradient(colors: [pColor, sColor], begin: Alignment.topLeft, end: Alignment.bottomRight)),
              child: Align(
                alignment: Alignment.bottomLeft,
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    if (lUrl.isNotEmpty) Image.network(lUrl, height: 30, errorBuilder: (c, e, s) => const SizedBox.shrink()),
                    const SizedBox(height: 12),
                    Text('Olá, ${widget.content.toUpperCase()}', style: const TextStyle(color: Colors.white, fontSize: 20, fontWeight: FontWeight.bold)),
                    const SizedBox(height: 4),
                    Text('Plano: ${widget.plano}', style: TextStyle(color: Colors.white.withOpacity(0.9), fontSize: 13)),
                  ],
                ),
              ),
            ),
            Expanded(
              child: ListView(
                padding: EdgeInsets.zero,
                children: [
                  _buildDrawerItem(icon: Icons.home_outlined, label: 'Início', onTap: () { _onItemTapped(0); Navigator.pop(context); }, primaryColor: pColor),
                  if (consumoIndex != -1) _buildDrawerItem(icon: Icons.data_usage_outlined, label: 'Consumo', onTap: () { _onItemTapped(consumoIndex); Navigator.pop(context); }, primaryColor: pColor),
                  if (suporteIndex != -1) _buildDrawerItem(icon: Icons.support_agent_outlined, label: 'Suporte', onTap: () { _onItemTapped(suporteIndex); Navigator.pop(context); }, primaryColor: pColor),
                  const Divider(height: 1, color: Color(0xFFE0E0E0), indent: 20, endIndent: 20),
                  if (mItems.isEmpty) const ListTile(title: Text("Nenhum item de menu configurado.", style: TextStyle(color: Colors.grey)))
                  else ...mItems.map((item) => _buildDrawerItem(icon: item.icon, label: item.name, onTap: () => _handleMenuNavigation(item), primaryColor: pColor)).toList(),
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildDrawerItem({required IconData icon, required String label, required VoidCallback onTap, required Color primaryColor}) {
    return ListTile(
      leading: Icon(icon, color: primaryColor),
      title: Text(label, style: const TextStyle(color: Color(0xFF333333), fontWeight: FontWeight.w500, fontSize: 15)),
      onTap: onTap,
      hoverColor: primaryColor.withOpacity(0.1),
      contentPadding: const EdgeInsets.symmetric(horizontal: 24, vertical: 4),
    );
  }
}

class InicioTab extends StatelessWidget {
  final String cpfCnpj, senha, content, plano, status, valorFatura, vencimentoFatura;
  final List<MenuItem> menuItems;
  final Map<String, dynamic> providerConfig;

  const InicioTab({
    Key? key,
    required this.cpfCnpj, required this.senha, required this.content, required this.plano,
    required this.status, required this.valorFatura, required this.vencimentoFatura,
    required this.menuItems, required this.providerConfig
  }) : super(key: key);

  void _handleGridNavigation(BuildContext context, MenuItem item) async {
    if (item.type == 'external_link' && item.url != null) {
      final uri = Uri.parse(item.url!);
      if (await canLaunchUrl(uri)) {
        await launchUrl(uri, mode: LaunchMode.externalApplication);
      } else {
        ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('Não foi possível abrir o link: ${item.url}')));
      }
    } else {
      navigateToPageById(
        context: context,
        pageId: item.id,
        cpfCnpj: cpfCnpj,
        senha: senha,
        status: status,
        providerConfig: providerConfig,
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    final configMap = providerConfig['config'] as Map<String, dynamic>? ?? {};
    final funcionalidades = configMap['features'] as Map<String, dynamic>? ?? {};
    final primaryColor = hexToColor(configMap['themeColor'] as String? ?? '#673AB7');
    final secondaryColor = hexToColor(configMap['secondaryColor'] as String? ?? '#9575CD');
    final carouselImages = (configMap['carousel'] as List<dynamic>? ?? []).map((item) => (item as Map)['url'].toString()).toList();

    return Stack(
      children: [
        _buildHeader(context, primaryColor, secondaryColor),
        Padding(
          padding: const EdgeInsets.fromLTRB(16.0, 160.0, 16.0, 0),
          child: ListView(
            padding: EdgeInsets.zero,
            children: [
              _buildStatusCard(context, primaryColor),
              const SizedBox(height: 16),
              if (funcionalidades['imageCarousel'] as bool? ?? false)
                Padding(
                  padding: const EdgeInsets.only(bottom: 16.0),
                  child: ImageSlider(imageUrls: carouselImages),
                ),
              if (funcionalidades['invoices'] as bool? ?? false)
                _buildInvoiceSection(context, primaryColor, secondaryColor),
              const SizedBox(height: 16),
              _buildServiceGrid(context, primaryColor),
              const SizedBox(height: 16),
              if ((configMap['tips'] as List<dynamic>? ?? []).isNotEmpty)
                Padding(
                  padding: const EdgeInsets.only(bottom: 16.0),
                  child: DicasUteisCard(providerConfig: providerConfig),
                ),
              if (funcionalidades['unlock'] as bool? ?? false)
                _buildDesbloqueioButton(context),
              const SizedBox(height: 120),
            ],
          ),
        ),
      ],
    );
  }

  Widget _buildServiceGrid(BuildContext context, Color primaryColor) {
    // ===== CORREÇÃO APLICADA AQUI =====
    // O filtro agora remove da grade apenas os itens que já estão em destaque na barra inferior.
    final gridItems = menuItems.where((item) =>
    item.id != 'home' &&
        item.id != 'internet_usage' && // Corresponde à aba 'CONSUMO'
        item.id != 'support'          // Corresponde à aba 'SUPORTE'
    ).toList();
    // ===================================

    if (gridItems.isEmpty) return const Center(child: Padding(padding: EdgeInsets.all(24.0), child: Text("Nenhuma ferramenta rápida configurada.", style: TextStyle(color: Colors.grey))));

    return GridView.builder(
      shrinkWrap: true,
      physics: const NeverScrollableScrollPhysics(),
      padding: EdgeInsets.zero,
      gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(crossAxisCount: 3, crossAxisSpacing: 12, mainAxisSpacing: 12, childAspectRatio: 1.0),
      itemCount: gridItems.length,
      itemBuilder: (context, index) {
        final service = gridItems[index];
        return _buildServiceItem(label: service.name, icon: service.icon, iconColor: primaryColor, onTap: () => _handleGridNavigation(context, service));
      },
    );
  }

  Widget _buildHeader(BuildContext context, Color primaryColor, Color secondaryColor) {
    return Container(

      decoration: BoxDecoration(gradient: LinearGradient(colors: [primaryColor, secondaryColor], begin: Alignment.topLeft, end: Alignment.bottomRight)),
      child: SafeArea(
        bottom: false,
        child: Padding(
          padding: const EdgeInsets.symmetric(horizontal: 10.0),
          child: Column(
            children: [
              const SizedBox(height: 16),

class _PromessaPageState extends State<PromessaPage> {
  PromiseState _currentState = PromiseState.eligible;
  String _message = '';
  late final Color primaryColor;

  @override
  void initState() {
    super.initState();
    final configMap = widget.providerConfig['config'] as Map<String, dynamic>? ?? {};
    primaryColor = hexToColor(configMap['themeColor'] as String? ?? '#673AB7');

    final bool isEligible = widget.status == 'Suspenso' || widget.status == 'Ativo V. Reduzida';
    if (!isEligible) {
      _currentState = PromiseState.notEligible;
      _message = "Esta opção está disponível apenas para contratos suspensos ou com velocidade reduzida.";
    }
  }

  Future<void> _makePromise() async {
    setState(() {
      _currentState = PromiseState.loading;
    });

    const apiUrl = 'http://45.176.56.70:3000/make-payment-promise';

    try {
      final details = widget.providerConfig['details'] as Map<String, dynamic>? ?? {};
      final sgpParams = {
        "token": details['apiToken'] as String? ?? '',
        "app": details['appName'] as String? ?? ''
      };

      if ((sgpParams['token'] ?? '').isEmpty || (sgpParams['app'] ?? '').isEmpty) {
        throw Exception("Token ou App Name não configurados no painel.");
      }

      final response = await http.post(
        Uri.parse(apiUrl),
        headers: {'Content-Type': 'application/json'},
        body: json.encode({
          'cpfCnpj': widget.cpfCnpj,
          'sgpParams': sgpParams,
        }),
      );

      if (response.statusCode == 200) {
        final responseData = json.decode(response.body);
        setState(() {
          _currentState = PromiseState.success;
          _message = responseData['message'] ?? 'Promessa realizada com sucesso!';
        });
      } else {
        final errorBody = json.decode(response.body)['error'];
        setState(() {
          _currentState = PromiseState.error;
          _message = errorBody['message'] ?? 'Ocorreu um erro desconhecido.';
        });
      }
    } catch (e) {
      setState(() {
        _currentState = PromiseState.error;
        _message = 'Erro de conexão: ${e.toString()}';
      });
    }
  }

    final order = menuData['order'] as List<dynamic>? ?? [];
    final items = menuData['items'] as Map<String, dynamic>? ?? {};
    final List<MenuItem> enabledItems = [];
    for (var id in order) {
      final itemData = items[id.toString()] as Map<String, dynamic>?;
      if (itemData != null && (itemData['enabled'] as bool? ?? false)) {
        final type = itemData['type'] as String? ?? 'internal';

        enabledItems.add(MenuItem(
          id: id.toString(),
          name: itemData['name'] as String? ?? pageNames[id.toString()] ?? 'Menu',
          type: type,
          url: itemData['url'] as String?,
          icon: pageIcons[type == 'external_link' ? 'external_link' : id.toString()] ?? pageIcons['default']!,
        ));
      }
    }
    return enabledItems;

import 'package:flutter/material.dart';

Color hexToColor(String hexString) {
  try {
    final buffer = StringBuffer();
    if (hexString.length >= 6) buffer.write('ff');

    // MUDANÇA AQUI: Adicionado .trim() para remover espaços
    buffer.write(hexString.trim().replaceAll('#', ''));

    return Color(int.parse(buffer.toString(), radix: 16));
  } catch (e) {
    // Retorna uma cor padrão em caso de erro de formatação
    print('Erro ao converter hex para cor: $e, String: "$hexString"');
    return const Color(0xFF673AB7);
  }
}
